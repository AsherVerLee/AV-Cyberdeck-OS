#dtparam=spi=on

# Enable audio (loads snd_bcm2835)
dtparam=audio=on

# Additional overlays and parameters are documented
# /boot/firmware/overlays/README

# Automatically load overlays for detected cameras
camera_auto_detect=1

# Automatically load overlays for detected DSI displays
display_auto_detect=1

# Automatically load initramfs files, if found
auto_initramfs=1

# Enable DRM VC4 V3D driver
dtoverlay=vc4-kms-v3d
max_framebuffers=2

# Don't have the firmware create an initial video= setting in cmdline.txt.
# Use the kernel's default instead.
disable_fw_kms_setup=1

# Run in 64-bit mode
arm_64bit=1

# Disable compensation for displays with overscan
disable_overscan=1

# Run as fast as firmware / board allows
arm_boost=1

# Enable PCIe for the future NVMe HAT. Harmless no-op until the HAT is
# physically attached - the controller just stays idle.
dtparam=pciex1

# Physical power button: momentary switch across GPIO3 (pin 5) and GND
# (pin 6). Press to shut down cleanly; press again to power back on.
dtoverlay=gpio-shutdown

# More aggressive Active Cooler fan curve than the firmware default -
# ramps up earlier/harder since it's sealed in a case with less airflow
# than an open board. Temps in millidegrees C, speed 0-255.
dtparam=cooling_fan=on
dtparam=fan_temp0=45000
dtparam=fan_temp0_hyst=2500
dtparam=fan_temp0_speed=60
dtparam=fan_temp1=52000
dtparam=fan_temp1_hyst=2500
dtparam=fan_temp1_speed=120
dtparam=fan_temp2=60000
dtparam=fan_temp2_hyst=2500
dtparam=fan_temp2_speed=180
dtparam=fan_temp3=68000
dtparam=fan_temp3_hyst=2500
dtparam=fan_temp3_speed=255

# WS2812/NeoPixel support over PIO for the 5 status LED buttons. Left
# disabled until wiring is finalized - with no params this claims GPIO4
# for a phantom 60-LED strip regardless of what's actually connected.
# Once wired, enable with real params, e.g.:
#dtoverlay=ws2812-pio,gpio=<pin>,leds=5

[cm4]
# Enable host mode on the 2711 built-in XHCI USB controller.
# This line should be removed if the legacy DWC2 controller is required
# (e.g. for USB device mode) or if USB support is not required.
otg_mode=1

[cm5]
dtoverlay=dwc2,dr_mode=host

[pi5]
dtoverlay=nospi10

[all]
